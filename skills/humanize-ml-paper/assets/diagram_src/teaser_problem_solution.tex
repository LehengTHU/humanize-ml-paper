\newcommand\poolrow[7]{%
  \node[#3] (#1t#2) at ([yshift={-(#2-1)*18.5pt}]#1o) {$x_{#2}$};
  \node[note, text=#5, anchor=west, text width=55pt, align=left] (#1q#2) at ([xshift=3pt]#1t#2.east) {\hdquote{#4}};
  \fill[score bar] ([xshift=3pt, yshift=-2.6pt]#1q#2.east) rectangle ++({#7*32pt}, 5.2pt);
  \ifdim #7pt<#6pt
    \path[penalty bar] ([xshift={3pt+#7*32pt}, yshift=-2.6pt]#1q#2.east) rectangle ++({(#6-#7)*32pt}, 5.2pt);
  \fi
}

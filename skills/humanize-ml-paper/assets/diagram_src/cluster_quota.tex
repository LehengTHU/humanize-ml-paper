% cluster_quota -- method figure at \columnwidth, on invented content: how the diversity scorer h_psi of
% framework_pipeline works. The encoder groups the candidates; group c gets the quota
% q_c = n_c^{-alpha} / sum_k n_k^{1-alpha} (alpha = 0.5: 0.8x, 1.0x, 1.2x, 1.7x of uniform for n = 5, 3, 2, 1),
% so the lone item of the rare group C_4 is sampled with p proportional to .8 x 1.7.
% Roles: rose = trainable, navy tint = quota, the boosted rare group is the one filled object.
\documentclass[tikz,border=1pt]{standalone}
\usepackage{housediagram}

% \grouprow{row}{n}{comma-separated item scores}{quota}{quota label}
\newcommand\grouprow[5]{%
  \coordinate (y#1) at (0, {-(#1-1)*21.5pt});
  \node[note, anchor=west] (la#1) at (y#1) {$C_{#1}$};
  \ifnum#1=4 \def\hdtok{token selected}\else\def\hdtok{token}\fi
  \foreach \w [count=\j] in {#3}{%
    \node[\hdtok, minimum width=14pt, minimum height=14pt] (a#1\j) at ([xshift={20pt + (\j-1)*16.5pt}]y#1) {\w};
  }
  \hdlane[inner ysep=2.5pt, inner xsep=4pt]{ra#1}{(la#1)([xshift=93pt, yshift=7pt]y#1)([yshift=-7pt]y#1)}
  \node[note, anchor=west] (lb#1) at ([xshift=125pt]y#1) {$n{=}#2$};
  \ifnum#1=4
    \fill[hdnavyhero] ([xshift=144pt, yshift=-3.5pt]y#1) rectangle ++({#4*26pt}, 7pt);
  \else
    \filldraw[fill=hdnavytint, draw=hdnavy, line width=0.8pt] ([xshift=144pt, yshift=-3.5pt]y#1) rectangle ++({#4*26pt}, 7pt);
  \fi
  \node[note, anchor=west] (qv#1) at ([xshift=190pt]y#1) {#5};
  \hdlane[inner ysep=2.5pt, inner xsep=4pt]{rb#1}{(lb#1)([xshift=205pt, yshift=7pt]y#1)([xshift=125pt, yshift=-7pt]y#1)}
}

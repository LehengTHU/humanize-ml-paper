% One frozen score ranks two copies of an easy question above a rare proof, so top-k keeps the copies

% Roles: teal outline = useful item, gray = noise, navy = frozen (the focal module, filled), coral = failure.

% \item{row}{outline colour}{quote}{score}
\newcommand\poolitem[4]{%
  \node[chip, draw=#2, tagged=#1, text width=62pt] (c#1) at (0, {-(#1-1)*18.5pt}) {\hdquote{#3}};
  \fill[score bar] ([xshift=3pt, yshift=-2.6pt]c#1.east) rectangle ++({#4*15pt}, 5.2pt);
}
